\documentclass[tikz,border=2pt]{standalone}
\usepackage{pgfplots}
\pgfplotsset{compat=newest}

\begin{document}
\begin{tikzpicture}
\begin{axis}[
  width=8cm, height=6cm,
  view={40}{40},
  hide axis,
]
  % u runs around the strip; v controls its width. Increase samples to refine the mesh.
  \addplot3[
    surf, colormap/cool, shader=interp,
    domain=0:360, y domain=-1:1,
    samples=40, samples y=5,
  ]
    ({(1+y/2*cos(x/2))*cos(x)},
     {(1+y/2*cos(x/2))*sin(x)},
     {y/2*sin(x/2)});
  % The v=0 centerline lies on the surface.
  \addplot3[black, thick, domain=0:360, samples=50]
    ({cos(x)},{sin(x)},{0});
\end{axis}
\end{tikzpicture}
\end{document}
